zz_valyria_magic_dragon_magic.001 = {
	type = character_event
	window = zz_valyria_magic_character_event
	title = zz_valyria_magic_dragon_magic.001.t
	desc = zz_valyria_magic_dragon_magic.001.desc
	theme = university 
	override_background = {
		reference = zz_dragon_magic
	}
	lower_left_portrait = root
	lower_right_portrait = {
		character = scope:secondary_recipient
	}
	lower_center_portrait = scope:recipient

	immediate = {
		zz_valyria_dead_dragon_egg_list = yes

	}

	widget = {
		is_shown = {
			is_ai = no
		}
		gui = "event_window_artifact_selection_tall"
		container = "custom_widgets_container"
	}
	option = {
		name = zz_valyria_magic_dragon_magic.001.opt.a
		trigger = {
			scope:recipient = {
				any_character_artifact = {
					has_variable = dragon_egg
					has_variable = dud_egg 
					NOT = { has_variable = cradled_egg }
					NOT = { has_variable = pitted_egg }
				}
			}
		}
		hidden_effect = {
			var:zz_valyria_selected_artifact?= { 
				save_scope_as = target
			}
			zz_valyria_magic_cast_effect = { 
				SKILL = dragon_magic
				SPELL = zz_valyria_magic_dragon_magic_001
			}
		}
	}
	option = {
		name = zz_valyria_magic_dragon_magic.001.opt.z
	}
}
