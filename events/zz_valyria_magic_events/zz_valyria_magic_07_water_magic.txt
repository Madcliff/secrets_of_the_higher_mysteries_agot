zz_valyria_magic_water_magic.002 = {
	type = character_event
	window = zz_valyria_magic_character_event
	title = zz_valyria_magic_water_magic.002.t
	desc = zz_valyria_magic_water_magic.002.desc
	theme = university 
	override_background = {
		reference = zz_water_magic
	}
	lower_left_portrait = root
	lower_right_portrait = {
		character = scope:secondary_recipient
	}
	lower_center_portrait = scope:recipient

	immediate = {
		clear_variable_list = possible_armies
		remove_variable = selected_army
		scope:recipient = {
			every_army = {
				root = {
					add_to_variable_list = {
						name = possible_armies
						target = prev
					}
					if = {
						limit = {
							NOT = { has_variable = selected_army }
						}
						set_variable = { name = selected_army value = prev }
					}
				}
			}
		}
	}

	widget = {
		is_shown = {
			is_ai = no
		}
		gui = "event_window_army_selection"
		container = "custom_widgets_container"
	}
	option = {
		name = zz_valyria_magic_water_magic.002.opt.a
		hidden_effect = {
			var:selected_army ?= { 
				save_scope_as = target
			}
			zz_valyria_magic_cast_effect = { 
				SKILL = water_magic
				SPELL  = zz_valyria_magic_water_magic_002
			}
		}
	}
	option = {
		name = zz_valyria_magic_water_magic.002.opt.z
	}
}

zz_valyria_magic_water_magic.004 = {
	type = character_event
	window = zz_valyria_magic_character_event
	title = zz_valyria_magic_water_magic.004.t
	desc = zz_valyria_magic_water_magic.004.desc
	theme = university 
	override_background = {
		reference = zz_water_magic
	}
	lower_left_portrait = root
	lower_right_portrait = {
		character = scope:secondary_recipient
	}
	lower_center_portrait = scope:recipient

	immediate = {
		zz_valyria_water_magic_epidemic_list = yes
		zz_targeted_character_province_selection_list_effect = yes 
		zz_valyria_water_magic_intensity_list = yes
	}
	widgets = {
		widget = {
			is_shown = {
				is_ai = no
			}
			gui = "event_window_epidemic_selection"
			container = "custom_widgets_container"
		}
		widget = {
			is_shown = {
				is_ai = no
			}
			gui = "event_window_province_selection"
			container = "custom_widgets_container"
		}
		widget = {
			is_shown = {
			}
			gui = "event_window_flag_selection"
			container = "custom_widgets_container"
		}
	}
	option = {
		name = zz_valyria_magic_water_magic.004.opt.e
		trigger = {
			OR = {
				var:selected_flag = flag:water_magic_intensity_01
				var:selected_flag = flag:water_magic_intensity_02
				var:selected_flag = flag:water_magic_intensity_03
			}
			
		}
		hidden_effect = {			
			var:selected_epidemic ?= { 
				save_scope_as = epidemic
			}
			switch = {
				trigger = var:selected_flag
				flag:water_magic_intensity_01 = {
					save_scope_value_as = {
						name = option_01
						value = yes
					}
				}
				flag:water_magic_intensity_02 = {
					save_scope_value_as = {
						name = option_02
						value = yes
					}
					
				}
				flag:water_magic_intensity_03 = {
					save_scope_value_as = {
						name = option_03
						value = yes
					}
					
				}
			}
			hidden_effect = {
				var:selected_province ?= { 
					save_scope_as = target
				}
				zz_valyria_magic_cast_effect = { 
					SKILL = water_magic
					SPELL  = zz_valyria_magic_water_magic_004
				}	
			}
		}
	}

	option = {
		name = zz_valyria_magic_water_magic.004.opt.z
	}
}
