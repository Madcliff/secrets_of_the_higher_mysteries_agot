
mystery_cost = {
	value = 0
	add ?= var:manacost
	multiply = this.caster_proficiency
}

	value = 1
	subtract = {
		value ?= var:cast_counter
		divide = 2000
		max = 0.4
	} 
}

	value = 0

		limit = { 
			exists = var:zz_manacost

			value ?= var:zz_manacost
		}
	}
	min = 1
}
