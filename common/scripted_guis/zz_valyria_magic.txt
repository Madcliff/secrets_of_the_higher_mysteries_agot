

zz_valyria_cast_click = {
	scope = character
	is_shown = {
		zz_open_mage_window = yes
	}
	is_valid = {

	}
	effect = {
		save_scope_as = actor 
		scope:actor = {
			if = {
				limit = {
					NOT = { exists = var:zz_miscast_count }
				}
				set_variable = {
					name = zz_miscast_count
					value = 0
				}
			}
		}
		scope:opened_character = {
			save_scope_as = recipient
		}
		zz_ui_setup_effect = yes 
	}
}

zz_valyria_magic_caster = {
	scope = character
	saved_scopes = {
		zz_magic_caster
	}
	is_shown = {
		var:zz_magic_caster ?= scope:zz_magic_caster
	}
	
	effect = {
		save_scope_as = zz_magic_user 
		set_variable = {
			name = zz_magic_caster
			value = scope:zz_magic_caster
		}
		if = {
			limit = { NOT = { exists = scope:zz_magic_user.var:zz_miscast_count } }
			scope:zz_magic_user ?= {
				set_variable = {
					name = zz_miscast_count
					value = 0
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:zz_magic_caster.var:zz_miscast_count } }
			scope:zz_magic_caster ?= {
				set_variable = {
					name = zz_miscast_count
					value = 0
				}
			}
		}
		if = {
			limit = { NOT = { exists = scope:zz_magic_user.var:zz_miscast_count } }
			scope:zz_magic_user ?= {
				set_variable = {
					name = zz_miscast_count
					value = 0
				}
			}
		}
		if = {
			limit = {
				exists = scope:zz_magic_user.var:zz_magic_target
			}
			scope:zz_magic_caster ?= {
				set_variable = {
					name = zz_magic_target
					value = scope:zz_magic_user.var:zz_magic_target
				}
			}
		}
		zz_valyria_make_spell_list = {MAGIC_USER = zz_magic_user MAGIC_CASTER = zz_magic_caster}
	}
}

zz_valyria_magic_cast = {
	scope = character
	is_valid = {
		exists = var:zz_valyria_mystery
		var:zz_magic_caster = {
			zz_valyria_has_mana_to_cast = {
				MANACOST = root.var:zz_valyria_mystery.mystery_cost
			}
		}
		
	}
	effect = {
		save_scope_as = actor
		var:zz_magic_caster = {
			save_scope_as = secondary_recipient
		}
		var:zz_magic_target = { 
			save_scope_as = recipient
		}
		var:zz_valyria_mystery = {
			save_scope_as = zz_valyria_mystery
		}
		scope:secondary_recipient = { 
			set_variable = {
				name = zz_manacost 
				value = current_mystery_cost
			}
			zz_valyria_mana_spend = yes
		}
		root = {
			#remove_variable = zz_magic_ui
		}
		scope:zz_valyria_mystery.var:spell_school = {
			save_scope_as = spell_school
		}
		scope:zz_valyria_mystery.var:spell_number = {
			save_scope_as = spell_number
		}
		scope:zz_valyria_mystery.var:spell_name = {
			save_scope_as = spell_name
		}
		zz_spell_trigger_switch_effect = { SCOPE = scope:zz_valyria_mystery.var:spell_name  EFFECT = zz_valyria_mystery_casting_gui_effect }
	}
}
