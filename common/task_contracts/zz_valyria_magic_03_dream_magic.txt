zz_valyria_magic_dream_magic_066_contract = {
	group = zz_valyria_magic_dream_magic_prophecies
	icon = "gfx/interface/icons/point_of_interest_types/poi_valyria_magic.dds"

	travel = no
	use_diplomatic_range = yes

	weight = {
		# Standard weights.
		value = task_contract_weight_default_value
		add = laamp_contracts_weight_up_learning_value
		# Employer weights.
	}

	# Validity Triggers
	valid_to_create = {
		# Standard triggers.
		scope:employer.highest_held_title_tier = tier_county
	}
	valid_to_accept = {
		# Standard triggers.
		
		#scope:employer = { is_landed = yes }
	}
	valid_to_continue = {
		# Standard triggers.
		task_contract_taker = { 

		} 
	}
	valid_to_keep = {
		# Standard triggers.
		task_contract_taker = { 
		} 
	}

	# On_actions
	on_accepted = {
		task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_accept_contract" }
		# For the intro event.
		save_scope_as = task_contract
		task_contract_employer = { save_scope_as = employer }
		# Then we roll our first event..
		task_contract_taker ?= { 
			root.task_contract_location = {
				save_scope_as = task_contract_location
			}
			trigger_event = zz_valyria_magic_dream_magic_prophecy_task.0001 
		}
	}
	on_create = {
		scope:contract = {
			# Make our gold rewards static.
			set_variable = { 
				name = quest_reward_01
				value = zz_valyria_magic_dream_magic_066_value_01
			}
			set_variable = { 
				name = quest_reward_02
				value = zz_valyria_magic_dream_magic_066_value_02
			}
			set_variable = { 
				name = quest_reward_03
				value = zz_valyria_magic_dream_magic_066_value_03
			}
			set_variable = { 
				name = quest_reward_04
				value = zz_valyria_magic_dream_magic_066_value_04
			}
		}
	}

	# Rewards
	task_contract_reward = {
		success_standard = {
			should_print_on_complete = yes
			effect = {
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract" }
				task_contract_taker = { 
					if = {
						limit = {
							has_treasury = no 
						}
						add_gold = scope:contract.var:cash_reward_01
					}
					if = {
						limit = {
							has_treasury = yes 
						}
						add_gold = scope:contract.var:cash_reward_02
						add_treasury = scope:contract.var:cash_reward_03
					}
					if = {
						limit = { 
							domicile ?= {
								domicile_uses_provisions = yes 
							}
						}
						domicile ?= { change_provisions = scope:contract.var:cash_reward_04 }
					}
				}
				task_contract_taker = { task_successful_noncrim_contract_count_effect = yes }
			}
		}
		success_critical = {
			should_print_on_complete = yes
			effect = {
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_success_on_contract" }
				task_contract_taker = { 
					if = {
						limit = {
							has_treasury = no 
						}
						add_gold = scope:contract.var:quest_reward_01
					}
					if = {
						limit = {
							has_treasury = yes 
						}
						add_gold = scope:contract.var:quest_reward_02
						add_treasury = scope:contract.var:quest_reward_03
					}
					if = {
						limit = { 
							domicile ?= {
								domicile_uses_provisions = yes 
							}
						}
						domicile ?= { change_provisions = scope:contract.var:quest_reward_04 }
					}
				}
				task_contract_taker = { task_successful_noncrim_contract_count_effect = yes }
			}
		}
		failure_standard = {
			positive = no
			should_print_on_complete = yes
			effect = {
				task_contract_taker = { play_sound_effect = "event:/DLC/EP3/SFX/UI/Contracts/ep3_ui_contracts_failure_on_contract" }
			}
		}
	}
}
