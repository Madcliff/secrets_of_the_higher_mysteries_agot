zz_valyria_is_mage_ai = {
	OR = { 
		has_trait = zz_magister
		is_ai = yes 
		any_spouse = {
			has_trait = zz_magister 
			is_ai = yes 
		}
		house ?= { 
			any_house_mage = {
				exists = this 
				is_ai = yes 
			}
		}
		any_court_mage = {
			exists = this 
			is_ai = yes 
		}
		any_realm_mage = {
			exists = this 
			is_ai = yes 
		}
	}
}

zz_valyria_magic_is_ai_spouse_is_ai = {
	is_ai = yes  
	any_spouse = {
		NOT = {
			is_ai = no
		}
	}
}
zz_valyria_magic_is_not_player_court_mage = {
	has_court_position = zz_valyria_court_mages_court_position
	court_owner = {
		is_ai = yes 
	}
}
zz_valyria_ai_spell_cast_knows_spell_has_mana = {
	any_memory = {
		has_memory_type = $SPELL$
	}
	zz_valyria_has_mana_to_cast = {
		MANACOST = $SPELL$_cost
	}
}

zz_valyria_magic_valid_ai_caster = {
	has_trait = zz_magister
	zz_valyria_magic_is_ai_spouse_is_ai = yes 
	zz_valyria_magic_is_not_player_court_mage = yes
	zz_valyria_ai_spell_cast_knows_spell_has_mana = { SPELL = $SPELL$}
}

zz_valyria_mage_has_mana_knows_spell = {
	OR = {
		zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
		any_spouse_mage = {
			zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
		}			
		house ?= {
			any_house_mage = {
				zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
			}
		}
		any_court_mage = { 
			zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}

		}
		any_realm_mage = { 
			zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
		}
	}
}
