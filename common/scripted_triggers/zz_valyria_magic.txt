	OR = {
		has_character_modifier = zz_valyria_magic_drops_power

	}

zz_valyria_is_mage = {
	OR = { 
		has_trait = zz_magister
		any_spouse = {
			has_trait = zz_magister 
		}
		house ?= { 
			any_house_mage = {
				exists = this 
			}
		}
		any_court_mage = {
			exists = this 
		}
		any_realm_mage = {
			exists = this 
		}
	}
}



zz_valyria_can_become_apprentice = {
	OR = {
		not = {
			has_trait = zz_magister 
			has_trait = dragon 
		}
		
		
		any_spouse = {
			not = {
				has_trait = zz_magister
				has_trait = dragon 
			}
		}
		any_close_or_extended_family_member = {
			not = { 
				has_trait = zz_magister
				has_trait = dragon 
			}
		}
		house ?= { 
			any_house_member = {
				not = { 
					has_trait = zz_magister
					has_trait = dragon 
				}
			}
		}
		any_courtier_or_guest = { 
			not = { 
				has_trait = zz_magister
				has_trait = dragon 
			}
		}
		any_vassal_or_below = { 
			not = { 
				has_trait = zz_magister
				has_trait = dragon 
			}
		}
	}
}

zz_valyria_knows_spell = {
	any_memory = {
		has_memory_type = $SPELL$
	}		
}

zz_valyria_knows_spell_name = {
	save_temporary_scope_as = spell_known
	scope:spell_learner = {
		any_memory = {
			var:spell_name ?= scope:spell_known
		}	
	}
		
}

		}
	}
