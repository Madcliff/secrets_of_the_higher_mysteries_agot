zz_valyria_magic_dragon_magic_003_story = { #Used for the Scholarship progress event
	visible = yes 
	visualization = {
		character = {
			variable_name = "mystery_caster"
			label = "Caster"
		}
		character = {
			variable_name = "dragon_rider"
			label = "Dragon Rider"
		}
		basic_counter = {
			variable_name = "mystery_potency"
			label = "Mystery Strength"
			
		}
		basic_counter = {
			variable_name = "mystery_duration"
			label = "Years Remaining"
		}
	}
	on_setup = {
        story_owner = {
            var:caster = {
			    save_scope_as = secondary_recipient
		    }
		    remove_variable = caster
			var:actor = {
			    save_scope_as = actor
		    }
		    remove_variable = actor
			save_scope_as = dragon
		}
        set_variable = {
			name = mystery_potency
			value = zz_valyria_magic_dragon_magic_003_value
		}
		set_variable = {
			name = mystery_duration
			value = zz_valyria_magic_dragon_magic_003_value_02
		}
		round_variable = { name = mystery_duration nearest = 1 }
		set_variable = {
			name = mystery_caster
			value = scope:secondary_recipient
		}
		set_variable = {
			name = dragon_rider
			value = scope:dragon.var:current_rider
		}
	}

	on_end = {
		story_owner = {
		}
	}

	on_owner_death = {
		end_story = yes
	}

	effect_group = {
		days = 365
		first_valid = {
			triggered_effect = {
				trigger = {
					scope:story.var:mystery_duration > 0
				}
				effect = {
					change_variable = {
						name = mystery_duration
						subtract = 1
					}
					story_owner = {
						THIS = { 
							save_scope_as = dragon
						}
						send_interface_toast = {
							type = send_interface_message_as_feed_spell_buff_proc
							title = zz_blood_magic_draconic_enchantment
							desc = zz_blood_magic_draconic_enchantment.desc
							left_icon = scope:dragon
							add_prowess_skill = scope:story.var:mystery_potency 
							while = {
								count = scope:story.var:mystery_potency 
								change_dragon_size = {
									VALUE = 1			
								}
							}
							
						}
						if = { 
							limit = {
								has_variable = current_rider
							}
							var:current_rider ?= { 
								save_scope_as = rider 
							}
							scope:rider ?= {
								send_interface_toast = {
									type = send_interface_message_as_feed_spell_buff_proc
									title = zz_blood_magic_draconic_enchantment
									desc = zz_blood_magic_draconic_enchantment.desc
									left_icon = scope:rider
									right_icon = scope:dragon
								}
							}
						}
					}
				}
			}
			triggered_effect = {
				trigger = {
					scope:story.var:mystery_duration < 1
				}
				effect = {
					end_story = yes
				}
			}
		}

	}
}
