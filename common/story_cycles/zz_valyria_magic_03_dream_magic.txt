		first_valid = {
			triggered_effect = {
				trigger = {
					scope:story.var:mystery_duration > 0

										}

										}

										name = scheme	
										value = yes 
									} 
								}
								else = {
									random_spouse = {
										limit = {
											zz_magic_any_targeting_scheme = yes 
										}
										add_to_list = scrying_targets
									}
									random_close_family_member = {
										limit = {
											zz_magic_any_targeting_scheme = yes 
										}
										add_to_list = scrying_targets
									}

										limit = {	
											zz_magic_any_targeting_scheme = yes 
										}
										add_to_list = scrying_targets
									}

										limit = {	
											zz_magic_any_targeting_scheme = yes 
										}
										add_to_list = scrying_targets
									}

										limit = {	
											zz_magic_any_targeting_scheme = yes 
										}
										add_to_list = scrying_targets
									}
									random_councillor = {
										limit = {	
											zz_magic_any_targeting_scheme = yes 
										}
										if = {
											limit = {
												exists = scope:dreamer.primary_spouse
												scope:dreamer.primary_spouse = {
													zz_magic_any_targeting_scheme = yes 
												}
											}
											scope:dreamer.primary_spouse = {
												add_to_list = scrying_targets
											}
										}
										if = {
											limit = {
												exists = scope:dreamer.liege
												scope:dreamer.liege = {
													zz_magic_any_targeting_scheme = yes 
												}
											}
											scope:dreamer.liege = {
												add_to_list = scrying_targets
											}
										}			
									}
									random_in_list = {
										list = scrying_targets
										save_scope_as = scrying_target
									}
									if = {
										limit = {
											NOT = {	exists = scope:scrying_target	}
										}
										random_living_character = {
											limit = {
												zz_magic_any_targeting_scheme = yes 
											}
											save_scope_as = scrying_target
										}
										scope:scrying_target ?= {
											random_targeting_scheme = {
												limit = {
													zz_random_targeting_scheme = yes
												}
												save_scope_as = dreaming_magic_secret_target
											}
											
										}
										save_scope_value_as = {
											name = scheme
											value = yes 
										} 
									}

									else = {
										scope:scrying_target ?= {
											random_targeting_scheme = {
												limit = {
													zz_random_targeting_scheme = yes
												}
												if = {
													limit = { exists = this	}
													save_scope_as = dreaming_magic_secret_target
												}
											}
										}
										save_scope_value_as = {
											name = scheme
											value = yes 
										} 
									}
								}	
							}
							1 = {
								if = {
									limit = {
										any_targeting_secret = {
											exists = this
											NOT = {	is_known_by = scope:dreamer	}
										}
									}
									save_scope_as = scrying_target

										limit = {
											NOT = {	is_known_by = scope:dreamer	}
										}
										if = {
											limit = { exists = this	}
											save_scope_as = dreaming_magic_secret_target
											
										}
									}
									save_scope_value_as = {
										name = secret
										value = yes 
									} 
								}
								else = {
									random_spouse = {
										limit = {
											any_targeting_secret = {
												exists = this 
												NOT = {	is_known_by = scope:dreamer	}
											}
										}
										add_to_list = scrying_targets
									}
									random_close_family_member = {
										limit = {
											any_targeting_secret = {
												exists = this 
												NOT = {	is_known_by = scope:dreamer	}
											}
										}
										add_to_list = scrying_targets
									}
									random_neighboring_and_across_water_realm_same_rank_owner ={
										limit = {	
											has_any_secrets = yes
											any_secret = {
												NOT = {	is_known_by = scope:dreamer	}
											}
										}
										add_to_list = scrying_targets
									}

										limit = {	
											has_any_secrets = yes

											has_any_secrets = yes
											any_secret = {
												NOT = {	is_known_by = scope:dreamer	}
											}
										}
										add_to_list = scrying_targets
									}
									random_councillor = {
										limit = {	
											has_any_secrets = yes
											any_secret = {
												NOT = {	is_known_by = scope:dreamer	}
											}
										}

											limit = {

												add_to_list = scrying_targets
											}
										}
										if = {
											limit = {
												exists = scope:dreamer.liege
												scope:dreamer.liege = {
													has_any_secrets = yes
													any_secret = {
														NOT = {	is_known_by = scope:dreamer	}
													}
												}
											}
											scope:dreamer.liege = {
												add_to_list = scrying_targets
											}
										}			
									}

											random_secret = {
												limit = {
													NOT = {	is_known_by = scope:dreamer	}
												}
												save_scope_as = dreaming_magic_secret_target
											}
										}
										save_scope_value_as = {
											name = secret
											value = yes 
										} 
									}
									else = {
										scope:scrying_target = {
											random_secret = {
												limit = {
													NOT = {	is_known_by = scope:dreamer	}
												}
												if = {
													limit = { exists = this	}
													save_scope_as = dreaming_magic_secret_target
												}
											}
										}
										save_scope_value_as = {
											name = secret
											value = yes 
										} 
									}
								}
							}
						}

						if = {
							limit = {
								exists = scope:scrying_target
								exists = scope:dreaming_magic_secret_target
							}
							send_interface_toast = {
								type = send_interface_message_as_feed_spell_buff_proc
								title = zz_valyria_magic_dream_magic.004.buff.toast.t
								desc = zz_valyria_magic_dream_magic.004.buff.toast.desc

								right_icon = scope:scrying_target
								custom_tooltip = zz_valyria_magic_dream_magic.004.buff.toast.tt
								if = {
									limit = {
										scope:scheme ?= yes 
									}
									scope:dreaming_magic_secret_target = {
										#expose_scheme_to = scope:dreamer
										expose_scheme = yes
									}
								}
								if = {
									limit = {
										scope:secret ?= yes
									}
									scope:dreaming_magic_secret_target = {
										reveal_to = scope:dreamer
									}
								}
							}
						}
					}
				}
			}
			triggered_effect = {
				trigger = {
					scope:story.var:mystery_duration < 1
				}
				effect = {
					scope:story = {
						end_story = yes
					}
					
				}
			}
