zz_valyria_magic_education_magic_list = {
	remove_variable = selected_flag
	clear_variable_list = possible_flags
	add_to_variable_list = {
		name = possible_flags
		target = flag:$MAGIC_SCHOOL$_skill
	}
	set_variable = {
		name = selected_flag
		value = flag:$MAGIC_SCHOOL$_skill
	}
	every_in_global_list = {
		limit = {
			NOT = {
				ROOT = {
					zz_valyria_knows_spell_name = yes
				}
			}
		}
		variable = zz_valyria_$MAGIC_SCHOOL$_list
		add_to_variable_list = {
			name = possible_flags
			target = this
		}
	}
}
