zz_valyria_magic_event_target = {
	if = {
		limit = {
				$TRIGGER$ = yes
			}
			add_to_list = zz_mages_casting_target
		}
		else_if = {
			every_spouse = {
				limit = {
					$TRIGGER$ = yes
				}
				add_to_list = zz_mages_casting_target
			}
		}
		else_if = {
			every_close_family_member = {
				limit = {
					$TRIGGER$ = yes
				}
				add_to_list = zz_mages_casting_target
			}
		}

		save_scope_as = zz_magic_user
		random_in_list = {
			list = zz_mages_casting_target
			save_scope_as = zz_magic_target 
			save_scope_as = recipient
		}
}



#zz_valyria_magic_event_cast = { TARGET = SPELL = SPELLSCHOOL =  }
zz_valyria_magic_event_cast = {
	# $TARGET$
	# $SPELL$
	# $SPELLSCHOOL$
	save_scope_as = actor 
	save_scope_as = zz_magic_user
	if = {
		limit = { 
			zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
		}
		add_to_list = mages
	}
	every_spouse_mage = {
		limit = { 
			zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
		}
		add_to_list = mages
	}			
	house ?= {
		every_house_mage = {
			limit = { 
				zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
			}
			add_to_list = mages
		}
	}
	every_court_mage = { 
		limit = { 
			zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
		}
		add_to_list = mages
	}
	every_realm_mage = { 
		limit = { 
			zz_valyria_magic_valid_ai_caster = { SPELL = $SPELL$}
		}
		add_to_list = mages
	}
			
	random_in_list = {
		list = mages
		save_scope_as = zz_magic_caster
		save_scope_as = secondary_recipient
	}
	$TARGET$ = {
		save_scope_as = recipient
		save_scope_as = zz_magic_target
	}

		
	scope:secondary_recipient = { 
		random_memory = {
			limit = {
				has_memory_type = $SPELL$
			}
			save_scope_as = zz_valyria_mystery
		}
	}
	zz_valyria_magic_cast_effect = { 
		SKILL = $SPELLSCHOOL$
		SPELL = $SPELL$
	}
}
