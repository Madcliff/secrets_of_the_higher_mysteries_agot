				target = prev
			}
		}

			has_trait = zz_magister
		}
		scope:$MAGIC_USER$ = {
			add_to_variable_list = {
				name = mages_list
				target = prev
			}
		}
	}

		limit = {
			liege_or_court_owner ?= scope:$MAGIC_USER$
		}
		scope:$MAGIC_USER$ = {
			add_to_variable_list = {
				name = mages_list
				target = prev
			}
		}
	}
	every_court_mage = {
		scope:$MAGIC_USER$ = {
			add_to_variable_list = {
				name = mages_list
				target = prev
			}
		}
	}
	every_realm_mage = {
		limit = {
			liege_or_court_owner ?= scope:$MAGIC_USER$
		}
		scope:$MAGIC_USER$ = {
			add_to_variable_list = {
				name = mages_list
				target = prev
			}
		}
	}
	
	every_in_list = {
		variable = mages_list
		if = {
			limit = {
				not = {

zz_valyria_make_item_caster_list = {
	
	if = {
		limit = { 
			var:zz_item_caster ?= { 
				NOT = {
					any_character_artifact = {
						has_variable = spell_name
						var:item_type ?= flag:scroll
					}
				}
			}
		}
		remove_variable = zz_item_caster
	}
	clear_variable_list = item_caster_list
	
	if = { 
		limit = {
			any_character_artifact = {
				has_variable = spell_name
				var:item_type ?= flag:scroll
			}
		}
		scope:$MAGIC_USER$ = {
			add_to_variable_list = {
				name = item_caster_list
				target = prev
			}
		}
	}
	every_spouse = {
		limit = {
			any_character_artifact = {
				has_variable = spell_name
				var:item_type ?= flag:scroll
			}
		}
		scope:$MAGIC_USER$ = {
			add_to_variable_list = {
				name = item_caster_list
				target = prev
			}
		}
	}
	every_courtier_or_guest = {
		limit = {
			any_character_artifact = {
				has_variable = spell_name
				var:item_type ?= flag:scroll
			}
		}
		scope:$MAGIC_USER$ = {
			add_to_variable_list = {
				name = item_caster_list
				target = prev
			}
		}
	}
	every_in_list = {
		variable = item_caster_list
		if = {
			limit = {
				not = {
					has_variable = zz_miscast_count
				}
			}
			set_variable = {
				name = zz_miscast_count
				value = 0
			}
		}
	}
	if = { 
		limit = {
			NOT = {
				has_variable = zz_item_caster
			}
		}
		ordered_in_list = {
			variable = item_caster_list
			position = 0
			save_scope_as = zz_item_caster
		}
		if = {
			limit = {
				exists = scope:zz_item_caster
			}
			set_variable = {
				name = zz_item_caster
				value = scope:zz_item_caster
			}
		}
	}
}

			}
		}
	}
	else = {
		random = {
			chance = 50

				name = zz_miscast_count

		}

zz_valyria_mana_gain = {
	if =  {
		limit = {
			not = {
				has_variable = zz_drops_of_power
			}
		}
		set_variable = {
			name = zz_drops_of_power
			value = 0
		}
	}
	set_variable = { name = managain value = $MANAGAIN$}
	change_variable = {
		name = managain 
		multiply = this.zz_valyria_mana_gain_mult
	}
	
	
	while = {
		limit = {
			has_variable = zz_drops_of_power
			var:zz_drops_of_power < zz_valyria_mana_cap
		}
		count = mana_gain_addition
		change_variable = {
			name = zz_drops_of_power
			add = 1
		}

	}
	remove_variable = managain
}

zz_valyria_magic_cast_effect = { 
	
	if = {
		limit = {
			OR = {
				NOT = {
					exists = var:cast_type
				}
				var:cast_type = flag:mystery
			}
			
		}
		scope:secondary_recipient = {
			remove_variable = casttype
			zz_valyria_magic_cast_xp_gain = {
				SKILL = $SKILL$
			}
			zz_valyria_mystery_cast_increment = yes
			zz_valyria_cost_rebalance = {SPELL = zz_valyria_magic_$SKILL$_$SPELL$_cost}
		}
		random_list = { 
			40 = { 
				modifier = {
					add = zz_spellcast_value
				}
				zz_valyria_magic_$SKILL$_$SPELL$_effect = {SPELLSCHOOL = $SKILL$ SPELL = $SPELL$}
			}
			60 = { 
				zz_valyria_spell_failure_effect = yes
			}
		}
	}
	else = {
		if = { 
			limit = {
				var:cast_type = flag:scroll
			}
			scope:zz_valyria_scroll = {
				if = {
					limit = {
						var:item_type ?= flag:scroll
						var:spell_name = flag:zz_valyria_magic_$SKILL$_$SPELL$
					}
					scope:secondary_recipient = {
						remove_variable = casttype
						set_variable = {
							name = casttype 
							value = flag:scroll
							days = 1
						}
						remove_variable = cast_potency
						set_variable = {
							name = cast_potency 
							value = scope:zz_valyria_scroll.var:cast_potency
						}
					}
				}
				destroy_artifact = this
			}
			zz_valyria_magic_$SKILL$_$SPELL$_effect = {SPELLSCHOOL = $SKILL$ SPELL = $SPELL$}
			if = { 
				limit = {
					scope:zz_item_caster ?= {
						any_character_artifact = {
							has_variable = spell_name
						}
					}
				} 
				root = {
					zz_valyria_make_scroll_list = {MAGIC_USER = zz_magic_user ITEM_CASTER = zz_item_caster}
				}
			}
			else = {
				root = {
					zz_valyria_make_item_caster_list = {MAGIC_USER = zz_magic_user}
					zz_valyria_make_scroll_list = {MAGIC_USER = zz_magic_user ITEM_CASTER = zz_item_caster}
				}
			}
		}
	}
	
	root = {
		save_scope_as = actor 
		
	}
}

zz_valyria_set_spell_values = {
	set_variable = { 
		name = spell_school 
		value = flag:$SPELLSCHOOL$
	}
	set_variable = { 
		name = spell_number
		value = flag:$NUMBER$
	}
	zz_set_spell_name = {SPELLSCHOOL = $SPELLSCHOOL$ NUMBER = $NUMBER$} 
	zz_assign_spell_type = yes 
	set_variable = { 
		name = manacost
		value = $MP$
	}
	set_variable = {
		name = cast_counter
		value = 1
	}
}


zz_add_spell = {
	if = {
		limit = {
			NOT = { 
				zz_valyria_knows_spell = {
					SPELL = zz_valyria_magic_$SPELLSCHOOL$_magic_$NUMBER$
				}
			}
		}
		save_scope_as = spell_learner
		scope:spell_learner = {
			create_character_memory = {
				type = zz_valyria_magic_$SPELLSCHOOL$_magic_$NUMBER$
			}
			scope:new_memory ?= {
				set_variable = {
					name = learning_location
					value = scope:spell_learner.location
				}
				set_variable = {
					name = is_mystery
					value = yes
				}
				zz_valyria_set_spell_values = {MP = zz_valyria_magic_$SPELLSCHOOL$_magic_$NUMBER$_cost SPELLSCHOOL = $SPELLSCHOOL$ NUMBER = $NUMBER$ }
				save_scope_as = zz_spell
			}
			
			scope:activity ?= {
				add_activity_log_entry = {
					key = zz_valyria_magic_learn_spell
					score = 25
					tags = { good learning activity }
					character = root
					#Effect
					custom_tooltip = zz_valyria_magic_learn_spell_desc
				}
			}
		}
		send_interface_message = {
			type = send_interface_message_as_feed_spell_learning
			title = zz_valyria_magic_learn_spell_title
			left_icon = scope:spell_learner
			right_icon = scope:spell_learner
			custom_tooltip = zz_valyria_magic_learn_spell_desc
			if = { 
				limit = {
					not = {
						has_trait = zz_magister
					}
				}
				add_trait_force_tooltip = zz_magister
			}
		}		
	}
}

zz_valyria_magic_learning_compare_memory = {
	scope:zz_magic_master = {
		every_memory = {
			limit = {
				has_variable = spell_name
				save_temporary_scope_as = current_spell
				NOT = {
					scope:zz_magic_learner = {
						any_memory = {
							var:spell_name ?= scope:current_spell.var:spell_name
						}
					}
				}
			}
			scope:zz_magic_learner = {
				PREV.var:spell_name = {
					save_scope_as = random_spell
				}
				save_scope_value_as = {
					name = spell 
					value = yes
				}
				zz_spell_trigger_switch_effect = { SCOPE = scope:random_spell EFFECT = zz_add_spell_or_scroll } 
			}
		}
	}
}

zz_scroll_making_effect = {

	scope:newly_created_artifact ?= {
		set_variable = {
			name = item_type
			value = flag:scroll 
		}
		zz_valyria_set_scroll_values = {SPELLSCHOOL = $SPELLSCHOOL$ NUMBER = $NUMBER$ }
		if = {
			limit = {
				exists = scope:zz_scroll_maker
			}
			set_variable = {
				name = cast_potency
				value = {
					scope:zz_scroll_maker = {
						value = zz_valyria_higher_mysteries_stat_gui
					}
				}
			}
			set_variable = {
				name = creator
				value = scope:zz_scroll_maker
			}
			add_artifact_history = {
				type = created
				recipient = scope:owner
			}
		}
		else = {
			set_variable = {
				name = cast_potency
				value = $POTENCY$
			}
		}
		set_artifact_name = zz_valyria_artifact_magic_scroll_name
	}
}

		}

			zz_scroll_making_effect = { POTENCY = 50 SPELLSCHOOL = $MAGIC$ NUMBER = $NUMBER$}
		}
	}
}
