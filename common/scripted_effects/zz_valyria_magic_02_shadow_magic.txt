zz_valyria_magic_shadow_magic_001_effect = {
	random_list = {
		75 = {
			modifier = { 
				add = zz_valyria_magic_$SPELLSCHOOL$_$SPELL$_value 
			}
			zz_valyria_magic_spell_success_interface = { SPELLSCHOOL = $SPELLSCHOOL$ SPELL = $SPELL$}
			if = {
				limit = {
					NOT = {
						scope:actor = scope:recipient
					}
				}
				stress_impact = {
					compassionate = medium_stress_impact_loss
					sadistic = minor_stress_gain
					callous = minor_stress_gain
				}
			}
			scope:recipient = { 
				zz_valyria_magic_apply_character_modifier_effect = { 
					MODIFIER = zz_valyria_magic_shadow_magic_001_modifier
					DURATION = 2
					BOOSTED_DURATION = 5
				}
			}			
			scope:recipient = {		
				if = {
					limit = {
						NOT = { scope:actor = scope:recipient }
					}
					add_opinion = {
						modifier = grateful_opinion
						target = scope:actor
						opinion = 70
					}
				}
			}
		}
		25 = {
			zz_valyria_magic_spell_failure_interface = { SPELLSCHOOL = $SPELLSCHOOL$ SPELL = $SPELL$}
		}
	}
}
